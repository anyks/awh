331 // Добавляем входные аргументы функции

334 debug.append(this->serialization(params));
335 // Добавляем перенос строки

337 // Добавляем описание входящего сообщения
338 debug.append("\x1B[1mMessage:\x1B[0m"AWH_STRING_BREAK);
339 }
340 // Выводим полученный нами лог
341 this->print(debug + format, flag, args...);
342 // Выводим лог в том виде как он пришёл

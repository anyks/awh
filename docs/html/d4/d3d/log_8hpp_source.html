322 void debug(const string & format, const string & method, const tuple <T...> & params, flag_t flag, Args&&... args) const noexcept {
323 // Если формат строки вывода передан
324 if(!format.empty()){
325 // Если метод названия функции передан
326 if(!method.empty()){
327 // Формируем результирующую строку отладки
328 string debug = AWH_STRING_BREAKS"\x1B[1mCalled function:\x1B[0m"AWH_STRING_BREAK;
329 // Добавляем название метода
330 debug.append(method);
331 // Добавляем перенос строки
332 debug.append(AWH_STRING_BREAKS);
333 // Если аргументы функции переданы
334 if(this->count(params) > 0){
335 // Добавляем входные аргументы функции
336 debug.append("\x1B[1mArguments function:\x1B[0m"AWH_STRING_BREAK);
337 // Добавляем список аргументов функции
338 debug.append(this->serialization(params));
339 // Добавляем перенос строки
340 debug.append(AWH_STRING_BREAKS);
341 // Добавляем описание входящего сообщения
342 debug.append("\x1B[1mMessage:\x1B[0m"AWH_STRING_BREAK);
343 }
344 // Выводим полученный нами лог
345 this->print(debug + format, flag, args...);
346 // Выводим лог в том виде как он пришёл
